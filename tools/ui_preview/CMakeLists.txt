cmake_minimum_required(VERSION 3.16)
project(tire_pressure_preview C)

get_filename_component(APP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LVGL_DIR "${APP_ROOT}/.pio/libdeps/waveshare_lcd_7/lvgl" CACHE PATH "LVGL 9.5 source directory")
if(NOT EXISTS "${LVGL_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "LVGL missing. Run: pio pkg install -e waveshare_lcd_7, or set -DLVGL_DIR=/path/to/lvgl")
endif()
set(LV_BUILD_CONF_PATH "${APP_ROOT}/include/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_DIR}" "${CMAKE_BINARY_DIR}/lvgl")
add_subdirectory("${APP_ROOT}/src/UI" "${CMAKE_BINARY_DIR}/ui")
target_compile_definitions(lib-ui PUBLIC LV_LVGL_H_INCLUDE_SIMPLE)

add_executable(ui_preview main.c "${APP_ROOT}/src/app_ui.c")
target_include_directories(ui_preview PRIVATE "${APP_ROOT}/include")
target_link_libraries(ui_preview PRIVATE lib-ui lvgl m)
target_compile_options(ui_preview PRIVATE -Wall -Wextra -Werror)

enable_testing()
add_test(NAME dashboard_render COMMAND ui_preview "${CMAKE_BINARY_DIR}/dashboard.ppm")

add_executable(test_gt911 test_gt911.c "${APP_ROOT}/src/gt911_report.c")
target_include_directories(test_gt911 PRIVATE "${APP_ROOT}/include")
target_compile_options(test_gt911 PRIVATE -Wall -Wextra -Werror)
add_test(NAME gt911_reports COMMAND test_gt911)
